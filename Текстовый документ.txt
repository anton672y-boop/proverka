from flask import Flask, render_template, request, redirect, url_for, flash, jsonify
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime
import os

app = Flask(__name__)
app.config['SECRET_KEY'] = 'your-secret-key-change-me'
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///database.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)


# ==================== МОДЕЛИ ====================

class Track(db.Model):
    """Музыкальный трек с клипом"""
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200), nullable=False)
    artist = db.Column(db.String(200), nullable=False)
    genre = db.Column(db.String(100), default='Другое')
    description = db.Column(db.Text, default='')
    # Ссылки (YouTube embed / VK / прямые ссылки)
    music_url = db.Column(db.String(500), default='')  # аудио
    clip_url = db.Column(db.String(500), default='')   # видео-клип
    cover_url = db.Column(db.String(500), default='')
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

    ratings = db.relationship('Rating', backref='track', lazy=True, cascade='all, delete-orphan')

    @property
    def avg_rating(self):
        if not self.ratings:
            return 0
        return round(sum(r.score for r in self.ratings) / len(self.ratings), 1)

    @property
    def rating_count(self):
        return len(self.ratings)


class Rating(db.Model):
    """Оценка трека"""
    id = db.Column(db.Integer, primary_key=True)
    track_id = db.Column(db.Integer, db.ForeignKey('track.id'), nullable=False)
    score = db.Column(db.Integer, nullable=False)  # 1-10
    comment = db.Column(db.Text, default='')
    author = db.Column(db.String(100), default='Аноним')
    created_at = db.Column(db.DateTime, default=datetime.utcnow)


# ==================== РОУТЫ ====================

def embed_url(url: str) -> str:
    """Преобразует обычную ссылку YouTube в embed"""
    if not url:
        return ''
    if 'youtube.com/watch?v=' in url:
        vid = url.split('v=')[1].split('&')[0]
        return f'https://www.youtube.com/embed/{vid}'
    if 'youtu.be/' in url:
        vid = url.split('youtu.be/')[1].split('?')[0]
        return f'https://www.youtube.com/embed/{vid}'
    return url


@app.route('/')
def index():
    """Главная — список всех треков с клипами и оценками"""
    sort = request.args.get('sort', 'new')
    genre = request.args.get('genre', '')

    query = Track.query
    if genre:
        query = query.filter_by(genre=genre)

    tracks = query.all()

    if sort == 'rating':
        tracks.sort(key=lambda t: t.avg_rating, reverse=True)
    elif sort == 'title':
        tracks.sort(key=lambda t: t.title.lower())
    else:  # new
        tracks.sort(key=lambda t: t.created_at, reverse=True)

    genres = sorted({t.genre for t in Track.query.all()})

    return render_template('index.html', tracks=tracks, genres=genres,
                           current_sort=sort, current_genre=genre,
                           embed_url=embed_url)


@app.route('/track/<int:track_id>')
def track_detail(track_id):
    """Страница отдельного трека"""
    track = Track.query.get_or_404(track_id)
    return render_template('track.html', track=track, embed_url=embed_url)


@app.route('/add', methods=['GET', 'POST'])
def add_track():
    """Добавить трек с клипом"""
    if request.method == 'POST':
        track = Track(
            title=request.form['title'].strip(),
            artist=request.form['artist'].strip(),
            genre=request.form.get('genre', 'Другое').strip() or 'Другое',
            description=request.form.get('description', '').strip(),
            music_url=request.form.get('music_url', '').strip(),
            clip_url=request.form.get('clip_url', '').strip(),
            cover_url=request.form.get('cover_url', '').strip(),
        )
        if not track.title or not track.artist:
            flash('Название и исполнитель обязательны!', 'error')
            return redirect(url_for('add_track'))

        db.session.add(track)
        db.session.commit()
        flash('Трек добавлен!', 'success')
        return redirect(url_for('index'))

    return render_template('add.html')


@app.route('/rate/<int:track_id>', methods=['POST'])
def rate(track_id):
    """Оценить трек"""
    track = Track.query.get_or_404(track_id)
    try:
        score = int(request.form['score'])
    except (KeyError, ValueError):
        flash('Некорректная оценка', 'error')
        return redirect(url_for('track_detail', track_id=track_id))

    if not 1 <= score <= 10:
        flash('Оценка должна быть от 1 до 10', 'error')
        return redirect(url_for('track_detail', track_id=track_id))

    rating = Rating(
        track_id=track.id,
        score=score,
        comment=request.form.get('comment', '').strip(),
        author=request.form.get('author', '').strip() or 'Аноним',
    )
    db.session.add(rating)
    db.session.commit()
    flash('Спасибо за оценку!', 'success')
    return redirect(url_for('track_detail', track_id=track_id))


@app.route('/delete/<int:track_id>', methods=['POST'])
def delete_track(track_id):
    """Удалить трек"""
    track = Track.query.get_or_404(track_id)
    db.session.delete(track)
    db.session.commit()
    flash('Трек удалён', 'success')
    return redirect(url_for('index'))

@app.route('/api/tracks')
def api_tracks():
    """JSON-API со всеми треками"""
    tracks = Track.query.all()
    return jsonify([{
        'id': t.id,
        'title': t.title,
        'artist': t.artist,
        'genre': t.genre,
        'avg_rating': t.avg_rating,
        'rating_count': t.rating_count,
        'clip_url': t.clip_url,
        'music_url': t.music_url,
    } for t in tracks])


with app.app_context():
    db.create_all()
